- User data
    
    ```bash
    #!/bin/bash
    # 1. 更新系統並安裝 Apache
    dnf update -y
    dnf install -y httpd
    
    # 2. 啟動 Apache 並設定開機自啟動
    systemctl start httpd
    systemctl enable httpd
    
    # 3. 取得 EC2 Metadata (IMDSv2 模式)
    TOKEN=$(curl -X PUT "http://169.254.169.254/latest/api/token" -H "X-aws-ec2-metadata-token-ttl-seconds: 21600")
    INSTANCE_ID=$(curl -H "X-aws-ec2-metadata-token: $TOKEN" -s http://169.254.169.254/latest/meta-data/instance-id)
    AVAILABILITY_ZONE=$(curl -H "X-aws-ec2-metadata-token: $TOKEN" -s http://169.254.169.254/latest/meta-data/placement/availability-zone)
    PUBLIC_IP=$(curl -H "X-aws-ec2-metadata-token: $TOKEN" -s http://169.254.169.254/latest/meta-data/public-ipv4)
    
    # 4. 建立美觀的 index.html
    cat <<EOF > /var/www/html/index.html
    <!DOCTYPE html>
    <html lang="zh-TW">
    <head>
        <meta charset="UTF-8">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
        <title>Apache 部署成功</title>
        <script src="https://cdn.tailwindcss.com"></script>
        <link href="https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.0.0/css/all.min.css" rel="stylesheet">
    </head>
    <body class="bg-gray-100 min-h-screen flex items-center justify-center p-6">
        <div class="max-w-3xl w-full bg-white rounded-3xl shadow-2xl overflow-hidden">
            <div class="h-2 bg-blue-600"></div>
            
            <div class="p-8 md:p-12">
                <div class="flex items-center space-x-4 mb-8">
                    <div class="bg-blue-100 p-3 rounded-xl">
                        <i class="fa-solid fa-server text-3xl text-blue-600"></i>
                    </div>
                    <div>
                        <h1 class="text-3xl font-extrabold text-gray-900">Apache Web Server</h1>
                        <p class="text-blue-600 font-medium">已透過 User-data 自動化部署完成</p>
                    </div>
                </div>
    
                <div class="grid grid-cols-1 md:grid-cols-2 gap-4 mb-8">
                    <div class="border border-gray-200 rounded-2xl p-4">
                        <span class="text-xs font-bold text-gray-400 uppercase">Instance ID</span>
                        <p class="text-lg font-mono text-gray-800">${INSTANCE_ID}</p>
                    </div>
                    <div class="border border-gray-200 rounded-2xl p-4">
                        <span class="text-xs font-bold text-gray-400 uppercase">Availability Zone</span>
                        <p class="text-lg font-mono text-gray-800">${AVAILABILITY_ZONE}</p>
                    </div>
                    <div class="border border-gray-200 rounded-2xl p-4 md:col-span-2">
                        <span class="text-xs font-bold text-gray-400 uppercase">Public IPv4</span>
                        <p class="text-lg font-mono text-gray-800">${PUBLIC_IP}</p>
                    </div>
                </div>
    
                <div class="flex flex-wrap gap-2">
                    <span class="px-3 py-1 bg-green-100 text-green-700 rounded-full text-sm font-semibold">Running</span>
                    <span class="px-3 py-1 bg-gray-100 text-gray-700 rounded-full text-sm font-semibold">AL2023</span>
                    <span class="px-3 py-1 bg-gray-100 text-gray-700 rounded-full text-sm font-semibold">HTTPD</span>
                </div>
            </div>
    
            <div class="bg-gray-50 px-8 py-4 flex justify-between items-center border-t border-gray-100">
                <span class="text-gray-500 text-sm italic">Generated on $(date '+%Y-%m-%d %H:%M:%S')</span>
                <div class="flex space-x-2">
                    <div class="w-3 h-3 bg-red-400 rounded-full"></div>
                    <div class="w-3 h-3 bg-yellow-400 rounded-full"></div>
                    <div class="w-3 h-3 bg-green-400 rounded-full"></div>
                </div>
            </div>
        </div>
    </body>
    </html>
    EOF
    ```
